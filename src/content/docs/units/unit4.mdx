referencias, ciclo de vida de objetos y la diferencia entre C++ y C#. En esta unidad

std::vector: el arreglo dinámico de C++ moderno
std::vector es la estructura de datos más utilizada en C++ moderno. Internamente

```cpp
#include <iostream>
```

### Actividad 05
Vectores: copia vs referencia
En la unidad anterior exploraste cómo se comportan los objetos al copiarlos.
Ahora veamos qué ocurre con un vector.

```cpp
#include <iostream>
#include <vector>
using namespace std;

void modificarPorValor(vector<int> vec) {
    vec.push_back(999);
    vec[0] = -1;
    cout << "Dentro de modificarPorValor:" << endl;
    cout << "  data: " << vec.data() << ", size: " << vec.size() << endl;
}

void modificarPorReferencia(vector<int>& vec) {
    vec.push_back(999);
    vec[0] = -1;
    cout << "Dentro de modificarPorReferencia:" << endl;
    cout << "  data: " << vec.data() << ", size: " << vec.size() << endl;
}

int main() {
    vector<int> original = {10, 20, 30};
    cout << "Original - data: " << original.data()
         << ", size: " << original.size() << endl;

    cout << "\n--- Paso por valor ---" << endl;
    modificarPorValor(original);
    cout << "Después de paso por valor:" << endl;
    cout << "  original[0] = " << original[0]
         << ", size: " << original.size() << endl;

    cout << "\n--- Paso por referencia ---" << endl;
    modificarPorReferencia(original);
    cout << "Después de paso por referencia:" << endl;
    cout << "  original[0] = " << original[0]
         << ", size: " << original.size() << endl;

    return 0;
}
```

### Actividad 06
Lista enlazada: gestión manual de nodos en memoria
Ahora vamos a construir una estructura donde tú gestionas cada nodo de manera
manual. Esto te ayudará a entender cómo funcionan internamente muchas estructuras
de datos y reforzará tu dominio de punteros y memoria dinámica.

```cpp
#include <iostream>
using namespace std;

struct Nodo {
    int dato;
    Nodo* siguiente;

    Nodo(int _dato) : dato(_dato), siguiente(nullptr) {
        cout << "Nodo creado con dato: " << dato
             << " en dirección: " << this << endl;
    }

    ~Nodo() {
        cout << "Nodo destruido con dato: " << dato << endl;
    }
};

int main() {
    // Crear tres nodos en el heap
    Nodo* primero = new Nodo(10);
    Nodo* segundo = new Nodo(20);
    Nodo* tercero = new Nodo(30);

    // Enlazarlos
    primero->siguiente = segundo;
    segundo->siguiente = tercero;
    // tercero->siguiente ya es nullptr (fin de la lista)

    // Recorrer la lista
    cout << "\nRecorriendo la lista:" << endl;
    Nodo* actual = primero;
    while (actual != nullptr) {
        cout << "Dato: " << actual->dato
             << " | Dirección del nodo: " << actual
             << " | Siguiente: " << actual->siguiente << endl;
        actual = actual->siguiente;
    }

    // Inspecciona con el depurador:
    // - ¿Dónde están primero, segundo, tercero (los punteros)?
    // - ¿Dónde están los nodos a los que apuntan?
    // - En Memory 1, escribe la dirección de primero y observa
    //   el campo 'dato' y el campo 'siguiente' en memoria.

    // Liberar memoria
    delete primero;
    delete segundo;
    delete tercero;

    return 0;
}
```

En el depurador, inspecciona primero, segundo y tercero. Observa las

En Memory 1, escribe primero y observa los bytes. Identifica dónde
termina el campo dato y dónde comienza el campo siguiente.
Responde estas preguntas:

¿Los nodos están contiguos en memoria como un arreglo? ¿Por qué sí o por qué no?
¿Qué ventaja tiene una lista enlazada sobre un arreglo para insertar un elemento
en el medio? ¿Y qué desventaja tiene para acceder al elemento en la posición i?
¿Qué sucedería si asignas segundo = nullptr; sin haber hecho delete segundo;?
¿Y si haces delete segundo; pero luego intentas acceder a primero->siguiente->dato?

Auditoría de código generado por IA
En esta actividad vas a experimentar con algo que ocurrirá constantemente en tu

La siguiente clase fue generada por una IA a quien se le pidió: "Crea una clase
en C++ que implemente una lista dinámica de enteros con operaciones de agregar,
insertar en posición, eliminar y buscar."

```cpp
#include <iostream>
using namespace std;

class ListaDinamica {
public:
    int* datos;
    int tam;
    int capacidad;

    ListaDinamica(int capInicial = 4) {
        capacidad = capInicial;
        tam = 0;
        datos = new int[capacidad];
        cout << "Lista creada. Capacidad: " << capacidad << endl;
    }

    void agregar(int valor) {
        if (tam == capacidad) {
            capacidad *= 2;
            int* nuevo = new int[capacidad];
            for (int i = 0; i < tam; i++) {
                nuevo[i] = datos[i];
            }
            datos = nuevo;
            cout << "Redimensionada. Nueva capacidad: " << capacidad << endl;
        }
        datos[tam] = valor;
        tam++;
    }

    void insertarEn(int posicion, int valor) {
        if (tam == capacidad) {
            capacidad *= 2;
            int* nuevo = new int[capacidad];
            for (int i = 0; i < tam; i++) {
                nuevo[i] = datos[i];
            }
            delete[] datos;
            datos = nuevo;
        }
        for (int i = tam; i > posicion; i--) {
            datos[i] = datos[i - 1];
        }
        datos[posicion] = valor;
        tam++;
    }

    int buscar(int valor) {
        for (int i = 0; i < tam; i++) {
            if (datos[i] == valor) return i;
        }
        return -1;
    }

    void eliminar(int posicion) {
        for (int i = posicion; i < tam - 1; i++) {
            datos[i] = datos[i + 1];
        }
        tam--;
    }

    void imprimir() {
        cout << "[";
        for (int i = 0; i < tam; i++) {
            cout << datos[i];
            if (i < tam - 1) cout << ", ";
        }
        cout << "] (size: " << tam << ", cap: " << capacidad << ")" << endl;
    }
};

int main() {
    ListaDinamica lista;
    lista.agregar(10);
    lista.agregar(20);
    lista.agregar(30);
    lista.imprimir();

    lista.insertarEn(1, 15);
    lista.imprimir();

    lista.eliminar(2);
    lista.imprimir();

    // ¿Qué pasa aquí?
    ListaDinamica copia = lista;
    copia.agregar(999);
    copia.imprimir();
    lista.imprimir();

    return 0;
}
```

Imagina que estás desarrollando un sistema de partículas para un efecto visual
en un videojuego. Cada frame se crean y destruyen cientos de partículas. Un colega

```cpp
#include <iostream>
using namespace std;

struct Particula {
    float x, y;
    float vx, vy;
    int vida;
};

class SistemaParticulas {
public:
    Particula** particulas;
    int cantidad;
    int capacidad;

    SistemaParticulas(int cap) : capacidad(cap), cantidad(0) {
        particulas = new Particula*[capacidad];
    }

    void agregar(float x, float y) {
        if (cantidad < capacidad) {
            particulas[cantidad] = new Particula{x, y, 0, -1, 60};
            cantidad++;
        }
    }

    void eliminarMuertas() {
        for (int i = 0; i < cantidad; i++) {
            if (particulas[i]->vida <= 0) {
                delete particulas[i];
                particulas[i] = particulas[cantidad - 1];
                cantidad--;
                i--;
            }
        }
    }

    void actualizar() {
        for (int i = 0; i < cantidad; i++) {
            particulas[i]->x += particulas[i]->vx;
            particulas[i]->y += particulas[i]->vy;
            particulas[i]->vida--;
        }
    }
};

int main() {
    SistemaParticulas sp(1000);
    for (int frame = 0; frame < 120; frame++) {
        sp.agregar(400, 300);
        sp.agregar(400, 300);
        sp.actualizar();
        sp.eliminarMuertas();
    }
    return 0;
}
```
